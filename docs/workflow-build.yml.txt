name: Build data and deploy site

on:
  workflow_dispatch:          # "Run workflow" button in the Actions tab
  push:                       # any merged change to the site or pipeline rebuilds and redeploys (~15 min)
    branches: [main]
    paths: ['site/**', 'scripts/**', 'requirements.txt', '.github/workflows/build.yml']
  schedule:
    - cron: '17 6 1,15 * *'   # 1st and 15th of every month, 06:17 UTC — picks up new call reports within ~2 weeks of release

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 180
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Cache downloaded source files
        uses: actions/cache@v4
        with:
          path: |
            raw/ncua
            raw/fdic
            raw/geo
          key: raw-${{ github.run_id }}
          restore-keys: raw-
      - run: pip install -r requirements.txt
      - name: Run pipeline
        run: bash scripts/run_all.sh
      - name: Drop files not needed for hosting
        run: rm -rf site/data/history_state site/artifact.html
      - uses: actions/configure-pages@v5
        with:
          enablement: true
      - uses: actions/upload-pages-artifact@v3
        with:
          path: site
  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
